function fanOpen(L){
  var fan=$("fan");if(!fan){fan=document.createElement("div");fan.id="fan";document.body.appendChild(fan)}
  var n=L.length,mid=(n-1)/2,S0=Math.min(1,(window.innerWidth*.5)/270,(window.innerHeight*.62)/405),tm=[],html="",b=0,rm=window.matchMedia&&matchMedia("(prefers-reduced-motion:reduce)").matches;
  L.forEach(function(c,i){if(c.ri>L[b].ri)b=i});var br=RS[L[b].ri];
  L.forEach(function(c,i){var an=c.an,ci=c.ci,ri=c.ri,r=RS[ri];
    var fr=LORE(cn(an.c[ci])[0],cn(an.c[ci])[1],an.i[ci],an.l,stars(ri+1),nm(an)+" \u00b7 #"+(ci+1<10?"0":"")+(ci+1)+"/"+(an.c.length<10?"0":"")+an.c.length,gem(r.k),r);
    html+='<div class="flip fc" data-i="'+i+'" style="--k:'+(i-mid)+';--s:'+S0+';--rc:'+r.c+';z-index:'+(10-Math.round(Math.abs(i-mid)*2))+'"><div class="in"><div class="face back"><img class="bk2" src="/cardback.webp?v=2" alt=""></div><div class="face front lore '+r.k+'" style="--rc:'+r.c+'">'+fr+'</div></div></div>'});
  fan.className="";fan.style.setProperty("--rc",br.c);
  fan.innerHTML='<div class="frays"></div>'+html+'<div class="fanhint" id="fanhint"></div><button class="fanok" id="fanok" type="button">'+T("ok")+'</button>';
  void fan.offsetWidth;fan.classList.add("on");
  var cs=[].slice.call(fan.querySelectorAll(".fc")),ok=$("fanok"),hint=$("fanhint"),K=rm?.01:1;
  hint.textContent=LANG==="en"?"Tap a card to bring it forward":"\u0627\u0636\u063a\u0637 \u0639\u0644\u0649 \u0643\u0627\u0631\u0637 \u0628\u0627\u0634 \u062a\u0637\u0644\u0639 \u0644\u0644\u0642\u062f\u0627\u0645";
  function at(ms,f){tm.push(setTimeout(f,ms*K))}
  function flip(i,big){var el=cs[i],rar=RS[L[i].ri].k;el.classList.add("open","hit");try{chime(rar)}catch(e){}fxBurst(fan,el,rar,big);at(900,function(){el.classList.remove("hit")})}
  cs.forEach(function(el,i){at(250+i*110,function(){el.classList.add("in")})});
  var T1=250+n*110+650;at(T1,function(){cs.forEach(function(el){el.classList.add("spread")})});
  var T2=T1+1000,ord=[];L.forEach(function(c,i){if(i!==b)ord.push(i)});ord.sort(function(x,y){return L[x].ri-L[y].ri||x-y});
  ord.forEach(function(i,j){at(T2+j*460,function(){flip(i,false)})});
  var T3=T2+ord.length*460+500,sus=[300,500,950,1500][L[b].ri]||500,hi=L[b].ri>=2;
  at(T3,function(){fan.classList.add("lit","hush");cs[b].classList.add("big");if(hi)cs[b].classList.add("shiver")});
  at(T3+sus,function(){cs[b].classList.remove("shiver");cs.forEach(function(x){x.classList.remove("pick")});cs[b].classList.add("pick");fan.classList.remove("hush");
    fxFlash(fan,br.c,hi?.95:.45,hi?900:500);flip(b,hi);
    if(hi&&!rm)fan.animate([{transform:"translate(0,0)"},{transform:"translate(-9px,6px)"},{transform:"translate(8px,-7px)"},{transform:"translate(-5px,4px)"},{transform:"translate(0,0)"}],{duration:520,easing:"ease-out"})});
  at(T3+sus+1100,function(){ok.classList.add("show");hint.classList.add("show")});
  cs.forEach(function(el){el.onclick=function(){if(!el.classList.contains("open"))return;var p=el.classList.contains("pick");cs.forEach(function(x){x.classList.remove("pick")});if(!p)el.classList.add("pick")}});
  ok.onclick=function(){tm.forEach(clearTimeout);fan.classList.remove("on","lit","hush");setTimeout(function(){fan.innerHTML=""},400);var l=L[L.length-1];lastCard={id:l.an.id,i:l.ci};if(lastCard){S.bg=lastCard;lastCard=null;save()}render()}}
function finishFan(sb,ov){
  var L=PEND,b=0;L.forEach(function(c,i){if(c.ri>L[b].ri)b=i});
  var r=RS[L[b].ri],hi=L[b].ri>=2,d=hi?800:0;ov.setAttribute("data-rar",r.k);ov.style.setProperty("--rc",r.c);
  go(450,function(){sb.classList.add("rise");sb.classList.add("charge")});
  go(1250+d,function(){sb.classList.remove("charge");fxFlash(ov,r.c,hi?.95:.5,700);sb.classList.add("out")});
  go(1500+d,function(){PEND=null;fanOpen(L);go(500,function(){ov.classList.remove("show","burst","done");sb.className="sb"})})}
